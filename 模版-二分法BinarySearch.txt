1. 在xxx(v)vv中，找第一个对条件成立的位置

public int binarySearch (int left, int right){
	int start = left, end = right;
	while (start + 1 < end) {
		int mid = (start + end) // 2
		if (nums[mid] >= target) {
			end = mid; # 满足条件，说明(v)在mid的[左]边，移动end
		} else {
			start = mid;
		}
	}
	### 这个部分和if里面的正好相反。如果这里是start先，那if(condition(mid))则是end先。
	if (nums(start) == target) return start;
	if (nums(end)   == target) return end;
	return -1
}

public int binarySearch (int left, int right){
	int start = left, end = right;
	while (start + 1 < end) {
		int mid = (start + end) // 2
		if (nums[mid] <= target) {
			start = mid; # 满足条件，说明(v)在mid的[右]边，移动start
		} else {
			end = mid;
		}
	}
	if (nums(end)   == target) return end;
	if (nums(start) == target) return start;
	return -1
}
